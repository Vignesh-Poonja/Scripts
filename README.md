SELECT 
    ROUND(SUM(TOTAL_SIZE)/1024/1024/1024,2) AS TOTAL_DISK_SIZE_GB,
    ROUND(SUM(USED_SIZE)/1024/1024/1024,2)  AS USED_DISK_SIZE_GB,
    ROUND((SUM(TOTAL_SIZE) - SUM(USED_SIZE))/1024/1024/1024,2) AS FREE_DISK_SIZE_GB
FROM M_VOLUME_FILES;



SELECT
    DATABASE_NAME       AS SID,
    HOST                AS HOSTNAME,
    ROUND(SUM(USED_SIZE)/1024/1024/1024,2)  AS USED_DISK_SIZE_GB,
    ROUND(SUM(TOTAL_SIZE)/1024/1024/1024,2) AS TOTAL_DISK_SIZE_GB
FROM M_VOLUME_FILES,
     M_DATABASE
WHERE M_VOLUME_FILES.HOST = M_DATABASE.HOST
GROUP BY DATABASE_NAME, HOST;
